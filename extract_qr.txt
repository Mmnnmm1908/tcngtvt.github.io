import os
import openpyxl
from openpyxl_images import get_images
import pandas as pd

excel_file = 'qr.20.08.xlsx'
output_dir = 'anh-qr'

if not os.path.exists(excel_file):
    print(f"Lỗi: Không tìm thấy file '{excel_file}' trong thư mục!")
else:
    # 1. Tạo thư mục chứa ảnh nếu chưa có
    if not os.path.exists(output_dir):
        os.makedirs(output_dir)

    # 2. Đọc file Excel bằng openpyxl để lấy thông tin văn bản
    wb = openpyxl.load_workbook(excel_file, data_only=True)
    sheet = wb.active

    # Tạo một bản đồ (dictionary) để tra cứu: hàng nào ứng với CCCD nào
    # Cột D là cột số 4 (Số căn cước)
    cccd_map = {}
    for row in range(2, sheet.max_row + 1):
        cell_val = sheet.cell(row=row, column=4).value
        if cell_val is not None:
            cccd_map[row] = str(cell_val).strip()

    print("Đang quét vị trí và tự động trích xuất toàn bộ ảnh QR...")
    
    # 3. Sử dụng openpyxl_images để quét và lấy toàn bộ ảnh gốc ra
    images_dict = get_images(excel_file)
    success_count = 0

    # Duyệt qua từng ảnh tìm thấy trong file Excel
    for cell_address, img_data in images_dict.items():
        # Phân tách ô (Ví dụ: "G2" -> cột G, hàng 2)
        # Lấy số hàng để đối chiếu với danh sách CCCD
        row_num = int(''.join(filter(str.isdigit, cell_address)))
        col_name = ''.join(filter(str.isalpha, cell_address)).upper()

        # Chúng ta chỉ xử lý nếu bức ảnh đó nằm ở cột G (Cột số tiền/Mã QR của bạn)
        if col_name == 'G' and row_num in cccd_map:
            cccd_name = cccd_map[row_num]
            
            # Đường dẫn lưu tệp ảnh mới lấy số CCCD làm tên
            output_path = os.path.join(output_dir, f"{cccd_name}.png")
            
            # Lưu dữ liệu ảnh gốc xuống ổ đĩa
            with open(output_path, "wb") as f:
                f.write(img_data)
                
            print(f"-> Đã trích xuất xong: {cccd_name}.png (Từ ô {cell_address})")
            success_count += 1

    print(f"\n Hoàn thành tuyệt đối! Đã tự động đổi tên {success_count} ảnh QR thành công vào thư mục '{output_dir}'.")
